cmake_minimum_required(VERSION 3.10)
project(DX11BusMultiplayer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
    message(FATAL_ERROR "Este projeto usa Direct3D 11 e requer Windows.")
endif()
add_executable(bus_server src/server.cpp)
target_link_libraries(bus_server ws2_32)
find_program(FXC_EXECUTABLE NAMES fxc fxc.exe)
if(NOT FXC_EXECUTABLE)
    message(FATAL_ERROR "fxc.exe nao encontrado. Passe -DFXC_EXECUTABLE=C:/caminho/fxc.exe")
endif()
set(VS_CSO "${CMAKE_CURRENT_BINARY_DIR}/bus_vs.cso")
set(PS_CSO "${CMAKE_CURRENT_BINARY_DIR}/bus_ps.cso")
add_custom_command(OUTPUT "${VS_CSO}" COMMAND "${FXC_EXECUTABLE}" /nologo /T vs_4_0 /E VSMain /Fo "${VS_CSO}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/bus.hlsl" DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/bus.hlsl" VERBATIM)
add_custom_command(OUTPUT "${PS_CSO}" COMMAND "${FXC_EXECUTABLE}" /nologo /T ps_4_0 /E PSMain /Fo "${PS_CSO}" "${CMAKE_CURRENT_SOURCE_DIR}/shaders/bus.hlsl" DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/shaders/bus.hlsl" VERBATIM)
add_custom_target(bus_shaders ALL DEPENDS "${VS_CSO}" "${PS_CSO}")
add_executable(bus_client WIN32 src/client.cpp)
add_dependencies(bus_client bus_shaders)
target_link_libraries(bus_client d3d11 dxgi ws2_32 user32 gdi32)
add_custom_command(TARGET bus_client POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${VS_CSO}" "$<TARGET_FILE_DIR:bus_client>/bus_vs.cso" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PS_CSO}" "$<TARGET_FILE_DIR:bus_client>/bus_ps.cso")
